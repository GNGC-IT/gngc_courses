```gdscript
extends RayCast3D

@export var hit_force: float = 15.0
@export var crosshair: ColorRect

@onready var infoLabel: Label = $"../../../../CanvasLayer/Label"

var blankColor: Color = Color("00000080") 
var highlightColor: Color = Color("FF000080")

var last_collider: Object = null

func _physics_process(_delta):
	var current_collider
	if is_colliding():
		current_collider  = get_collider()
	else:
		current_collider  = null
	
	# If the target changed, update the UI
	if current_collider != last_collider:
		last_collider = current_collider
		if current_collider:
			crosshair.color = highlightColor
			infoLabel.text = current_collider.name
		else:
			crosshair.color = blankColor
			infoLabel.text = ""

func _unhandled_input(event):
	if event.is_action_pressed("fire") and is_colliding():
		var body = get_collider()
		if body is RigidBody3D:
			apply_force(body)

func apply_force(body: RigidBody3D):
	var shoot_direction = -global_transform.basis.z
	var impulse = shoot_direction * hit_force
	var offset = get_collision_point() - body.global_position
	body.apply_impulse(impulse, offset)
```

The Godot Docs definition for "Apply Impulse"

>**apply_impulse**(impulse: [Vector3](https://docs.godotengine.org/en/stable/classes/class_vector3.html#class-vector3), position: [Vector3](https://docs.godotengine.org/en/stable/classes/class_vector3.html#class-vector3) = Vector3(0, 0, 0)) [](https://docs.godotengine.org/en/stable/classes/class_rigidbody3d.html#class-rigidbody3d-method-apply-impulse)
>Applies a positioned impulse to the body.
>
>An impulse is time-independent! Applying an impulse every frame would result in a framerate-dependent force. For this reason, it should only be used when simulating one-time impacts (use the "_force" functions otherwise).
>
>`position` is the offset from the body origin in global coordinates.
